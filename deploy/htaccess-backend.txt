# ---------------------------------------------------------------------------
# NovaNest - shared-hosting security snippet (OPTIONAL fallback)
#
# Only needed if you place the app under your web root (e.g. public_html).
# On Hostinger, prefer creating the Node.js app with an Application root that
# is NOT inside public_html (see HOSTINGER-SHARED.md) - that avoids exposing
# the database and uploads entirely.
#
# If the app must live under public_html, copy this file as `.htaccess` inside
# the app's backend/ folder to block web access to .env and data files.
# ---------------------------------------------------------------------------
<IfModule mod_rewrite.c>
    RewriteEngine On
    # Block direct web access to sensitive files
    RewriteRule ^\.env - [F,L]
    RewriteRule ^data/ - [F,L]
    RewriteRule ^\.env\.example$ - [F,L]
</IfModule>

<FilesMatch "^(\.env|\.env\.example|novanest\.json)$">
    Require all denied
</FilesMatch>
